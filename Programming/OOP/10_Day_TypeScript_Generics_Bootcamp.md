# DAY 4 — Generic Constraints

## Goals

***

# Problem

This fails:

```typescript
function printLength<T>(item: T): void {
  console.log(item.length);
}
```

TypeScript warns because not all types have:

```typescript
length
```

***

# Solution — Constraints

```typescript
interface HasLength {
  length: number;
}

function printLength<T extends HasLength>(
  item: T
): void {
  console.log(item.length);
}
```

***

# Usage

```typescript
printLength("hello");

printLength([1, 2, 3]);
```

# Generic Repository

## src/repositories/repository.ts

```typescript
export class Repository<T> {
  protected items: T[] = [];

  add(item: T): void {
    this.items.push(item);
  }

  getAll(): T[] {
    return this.items;
  }

  count(): number {
    return this.items.length;
  }
}
```

***

# Usage

```typescript
interface Weapon {
```

# Generic Save Service

## src/services/save-service.ts

```typescript
export class SaveService<T> {

  async save(
    filename: string,
    data: T
  ): Promise<void> {

    await Deno.writeTextFile(
      filename,
      JSON.stringify(data, null, 2)
    );
  }
}
```

***

# Usage


# Best Practices

## Use meaningful names

Good:

```typescript
<TItem>
<TData>
<TResponse>
```

Avoid:

```typescript
<TTT>
```

